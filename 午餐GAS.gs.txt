// ============================================================
// 午餐叫飯小幫手（獨立版）— Google Apps Script 後端
//
// 部署方式：
//   1. 建一個新的 Google Apps Script 專案（在 script.google.com 新增，
//      不要用同事那個既有的專案）
//   2. 貼上這整段程式碼，存檔
//   3. 左側「專案設定」齒輪 > 指令碼屬性(Script Properties)，新增：
//        GEMINI_API_KEY = 你在 aistudio.google.com 申請的金鑰
//   4. 上方函式下拉選單選 setupSpreadsheet，按執行（第一次會跳權限授權，
//      同意即可）。執行完在「執行紀錄」會看到新試算表的網址，記下來，
//      之後到那份試算表的「主名單表」手動建立初始名單（姓名 + 登入密碼欄填密碼）
//   5. 部署 > New deployment > Web App
//        - Execute as: 我 (Me)
//        - Who has access: 任何人 (Anyone)
//   6. 複製產生的網址，告訴 Claude 填入 Worker 的 GAS_URL
//
// 之後如果改了程式碼，要「部署 > 管理部署作業 > 編輯(鉛筆) > 新增版本」
// 才會生效，直接存檔不會更新已部署的網址內容。
// ============================================================

const MAIN_TAB = '主名單表';
const FLOW_TAB = '金流紀錄表';
const NAME_MAP_TAB = '姓名對照表'; // Uber顯示名稱 → 主名單表姓名，第一次用時會自動建立
const GEMINI_MODEL = 'gemini-3.6-flash'; // gemini-2.0-flash 已停用(2026-08 Google API 回應建議改用此型號)
const SCREENSHOT_DRIVE_FOLDER = '午餐叫飯小幫手截圖'; // 上傳的 Uber Eats 截圖存這個 Drive 資料夾，第一次用時會自動建立

function getSS_() {
  const id = PropertiesService.getScriptProperties().getProperty('SS_ID');
  if (!id) throw new Error('尚未設定試算表，請先在編輯器執行 setupSpreadsheet()');
  return SpreadsheetApp.openById(id);
}

// 一次性：建立全新試算表 + 主名單表/金流紀錄表兩分頁 + 表頭。
// 在 Apps Script 編輯器手動執行一次即可，不用每次部署都跑。
function setupSpreadsheet() {
  const props = PropertiesService.getScriptProperties();
  if (props.getProperty('SS_ID')) {
    Logger.log('已經建立過試算表了，網址：' + SpreadsheetApp.openById(props.getProperty('SS_ID')).getUrl());
    Logger.log('如果要重新建立一份新的，請先手動清除 Script Properties 裡的 SS_ID。');
    return;
  }

  const ss = SpreadsheetApp.create('午餐錢小幫手（獨立版）');
  const main = ss.getSheets()[0];
  main.setName(MAIN_TAB);
  main.getRange(1, 1, 1, 13).setValues([[
    '姓名', 'LINE_ID', '餘額', '上次通知日', '註冊時間', '角色',
    '共用錢包歸戶', '低額逾期狀態', '低額開始日', '最近提醒日', '登入密碼',
    '可儲值', '可輸入訂單'
  ]]);

  const flow = ss.insertSheet(FLOW_TAB);
  flow.getRange(1, 1, 1, 14).setValues([[
    '訂單標題', '姓名', '類型', '訂單金額', '外送費', '變動金額',
    '原餘額', '新餘額', '操作者', '備註', '時間戳記', '共用錢包歸戶', '打單日期', '截圖連結'
  ]]);

  props.setProperty('SS_ID', ss.getId());
  Logger.log('新試算表已建立：' + ss.getUrl());
  Logger.log('接下來請到「主名單表」手動建立初始名單（姓名 + 登入密碼欄填密碼，填了密碼才能登入；' +
    '可儲值/可輸入訂單這兩欄填 V 代表有權限，空白代表沒有）');
}

// 一次性：幫已經建立過的舊試算表補上「可儲值」「可輸入訂單」這兩個新欄位(L、M)。
// 如果是全新用 setupSpreadsheet() 建的試算表，已經內建這兩欄，不用跑這個。
// 在 Apps Script 編輯器手動執行一次即可，重複執行也安全(已經有標題就不會重複加)。
function addPermissionColumns() {
  const ss = getSS_();
  const sheet = ss.getSheetByName(MAIN_TAB);
  const headers = sheet.getRange(1, 1, 1, Math.max(sheet.getLastColumn(), 11)).getValues()[0];
  if (headers[11] === '可儲值' && headers[12] === '可輸入訂單') {
    Logger.log('已經補過這兩欄了，不用重複執行。');
    return;
  }
  sheet.getRange(1, 12, 1, 2).setValues([['可儲值', '可輸入訂單']]);
  Logger.log('已補上「可儲值」(L欄)、「可輸入訂單」(M欄)。請到主名單表幫需要的人填 V。');
}

// 一次性：幫已經建立過的舊試算表補上「截圖連結」這個新欄位(金流紀錄表 N 欄)。
// 在 Apps Script 編輯器手動執行一次即可，重複執行也安全。

}

// 同事那套系統的 GAS 網址，只用來「讀」他的名單，不會寫回去
const COLLEAGUE_GAS_URL = 'https://script.google.com/macros/s/AKfycbxHHqoEIBDI7YvlJk5kqvKUeE49j2fODvvYd8xZlqn1szll2mw546bJ4UBgU51nv4-0pg/exec';

// 一次性：從同事那套系統匯入資料到「自己這份」試算表。只呼叫同事 GAS 的 getList/getFlow（唯讀），
// 完全不會寫回同事那邊。在 Apps Script 編輯器手動執行一次即可。
//   1. 主名單表：同事有、自己沒有的人 → 新增；名字已經存在的人 → 只更新餘額(密碼/其他欄位不動)。
//      這部分可以重複執行，每次都會把餘額同步成同事那邊最新的數字。
//   2. 金流紀錄表：把同事的消費/儲值歷史記錄整批複製過來，讓行事曆看得到過去的紀錄。
//      這部分只在自己的金流紀錄表還是空的時候做一次，避免重複執行造成同一筆紀錄被複製兩份。
function importFromColleague() {
  const ss = getSS_();

  // ---- 1. 主名單表：新增缺的人、同步已存在的人的餘額 ----
  const listResp = UrlFetchApp.fetch(COLLEAGUE_GAS_URL + '?action=getList', { muteHttpExceptions: true });
  if (listResp.getResponseCode() !== 200) {
    Logger.log('讀取同事名單失敗：HTTP ' + listResp.getResponseCode() + '：' + listResp.getContentText().slice(0, 300));
    return;
  }
  var listData;
  try { listData = JSON.parse(listResp.getContentText()); }
  catch (e) { Logger.log('同事名單回傳的內容不是合法 JSON：' + listResp.getContentText().slice(0, 300)); return; }
  const colRows = listData.values || [];

  const mainSheet = ss.getSheetByName(MAIN_TAB);
  const lastMainRow = mainSheet.getLastRow();
  const existingRowByName = {}; // 姓名 → 在自己主名單表的實際列號
  if (lastMainRow >= 2) {
    mainSheet.getRange(2, 1, lastMainRow - 1, 1).getValues().forEach(function (r, i) {
      if (r[0]) existingRowByName[r[0]] = i + 2;
    });
  }

  const toAdd = [];
  let added = 0, updated = 0;
  colRows.forEach(function (r) {
    const name = r[0];
    if (!name) return;
    const existRow = existingRowByName[name];
    if (existRow) {
      mainSheet.getRange(existRow, 3).setValue(r[2] || 0); // 只同步餘額，密碼欄等其他欄位不動
      updated++;
    } else {
      toAdd.push([
        name, r[1] || '', r[2] || 0, '', '', r[5] || '', r[6] || '', '', '', '', '', '', ''
        // 姓名/LINE_ID/餘額/(上次通知日空)/(註冊時間空)/角色/共用錢包歸戶/(低額逾期狀態空)/(低額開始日空)/(最近提醒日空)/登入密碼(不匯入)/可儲值(不匯入)/可輸入訂單(不匯入)
      ]);
      added++;
    }
  });
  if (toAdd.length) {
    mainSheet.getRange(mainSheet.getLastRow() + 1, 1, toAdd.length, 13).setValues(toAdd);
  }
  Logger.log('主名單表：新增 ' + added + ' 人、同步餘額 ' + updated + ' 人。密碼欄不會從同事那邊帶過來，要登入的人請自行設定密碼。');

  // ---- 2. 金流紀錄表：只在自己這邊還是空的時候整批匯入，避免重複匯入 ----
  const flowSheet = ss.getSheetByName(FLOW_TAB);
  if (flowSheet.getLastRow() >= 2) {
    Logger.log('金流紀錄表已經有資料了，這次不會再匯入同事的消費/儲值歷史（避免重複）。' +
      '如果真的要重新匯入，請先手動清空金流紀錄表內容（保留表頭）再重新執行這個函式。');
    return;
  }

  const flowResp = UrlFetchApp.fetch(COLLEAGUE_GAS_URL + '?action=getFlow', { muteHttpExceptions: true });
  if (flowResp.getResponseCode() !== 200) {
    Logger.log('讀取同事消費紀錄失敗：HTTP ' + flowResp.getResponseCode() + '：' + flowResp.getContentText().slice(0, 300));
    return;
  }
  var flowData;
  try { flowData = JSON.parse(flowResp.getContentText()); }
  catch (e) { Logger.log('同事消費紀錄回傳的內容不是合法 JSON：' + flowResp.getContentText().slice(0, 300)); return; }
  const flowRows = flowData.values || [];
  if (!flowRows.length) { Logger.log('同事那邊沒有消費/儲值紀錄可以匯入'); return; }

  // 同事 getFlow 回的是顯示文字(getDisplayValues)，補齊/裁到 14 欄避免欄數不一致寫入失敗
  // (同事那邊沒有「截圖連結」這欄，補空白就好)
  const normalized = flowRows.map(function (r) {
    const row = r.slice(0, 14);
    while (row.length < 14) row.push('');
    return row;
  });
  flowSheet.getRange(flowSheet.getLastRow() + 1, 1, normalized.length, 14).setValues(normalized);
  Logger.log('金流紀錄表：匯入了同事那邊 ' + normalized.length + ' 筆歷史消費/儲值紀錄。');
}

function cleanName_(s) {
  return String(s || '').replace(/[.\s]/g, '');
}

// 讀「姓名對照表」，回傳 { 清洗過的Uber名字: 主名單表姓名 }；分頁不存在就回空物件
function getNameMap_(ss) {
  const sheet = ss.getSheetByName(NAME_MAP_TAB);
  const map = {};
  if (!sheet) return map;
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return map;
  const data = sheet.getRange(2, 1, lastRow - 1, 2).getValues();
  data.forEach(function (row) {
    const ub = cleanName_(row[0]);
    if (ub && row[1]) map[ub] = row[1]; // 同一個 Uber 名字有多筆對應時，後面的蓋過前面的
  });
  return map;
}

function getOrCreateNameMapSheet_(ss) {
  let sheet = ss.getSheetByName(NAME_MAP_TAB);
  if (!sheet) {
    sheet = ss.insertSheet(NAME_MAP_TAB);
    sheet.getRange(1, 1, 1, 2).setValues([['Uber顯示名稱', '對照姓名']]);
  }
  return sheet;
}

function doGet(e) {
  try {
    const action = e.parameter.action;
    const ss = getSS_();

    // --- 讀取名單 ---
    if (action === 'getList') {
      const sheet = ss.getSheetByName(MAIN_TAB);
      const lastRow = sheet.getLastRow();
      if (lastRow < 2) return respond_({ values: [] });
      const data = sheet.getRange(2, 1, lastRow - 1, 13).getValues();
      return respond_({ values: data.filter(function (row) { return row[0] !== ''; }) });
    }

    // --- 讀取金流紀錄（行事曆用）---
    if (action === 'getFlow') {
      const sheet = ss.getSheetByName(FLOW_TAB);
      const lastRow = sheet.getLastRow();
      const lastCol = Math.max(sheet.getLastColumn(), 14); // 至少讀到 N 欄（截圖連結）
      if (lastRow < 2) return respond_({ values: [] });
      const data = sheet.getRange(2, 1, lastRow - 1, lastCol).getDisplayValues();
      return respond_({ values: data.filter(function (row) { return row[0] !== '' || row[1] !== ''; }) });
    }

    // --- 送出扣款/儲值/修改 ---
    if (action === 'submit') {
      const payload = JSON.parse(e.parameter.data);
      const lock = LockService.getScriptLock();
      lock.waitLock(20000);
      try {
        const flowSheet = ss.getSheetByName(FLOW_TAB);
        const flowRows = payload.flowRows;
        if (flowRows.length > 0) {
          flowSheet
            .getRange(flowSheet.getLastRow() + 1, 1, flowRows.length, flowRows[0].length)
            .setValues(flowRows);
        }
        const mainSheet = ss.getSheetByName(MAIN_TAB);
        payload.balanceUpdates.forEach(function (u) {
          mainSheet.getRange(u.row, 3).setValue(u.value);
        });

        // 把這次匯入時人工確認過的「Uber名字→主名單表姓名」記下來，下次自動比對用
        const adds = payload.nameMapAdds;
        if (adds && adds.length) {
          const mapSheet = getOrCreateNameMapSheet_(ss);
          const rows = adds.map(function (x) { return [x.ubName, x.realName]; });
          mapSheet.getRange(mapSheet.getLastRow() + 1, 1, rows.length, 2).setValues(rows);
        }
      } finally {
        lock.releaseLock();
      }
      return respond_({ ok: true });
    }

    return respond_({ error: 'unknown action: ' + action });
  } catch (err) {
    return respond_({ error: err.message });
  }
}

// 截圖辨識：圖片資料量大，塞不進網址查詢字串，所以用真正的 POST
function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents);
    if (body.action === 'parseUberOrder') {
      const ss = getSS_();
      return respond_(parseUberOrder_(body.images || [], getNameMap_(ss)));
    }
    if (body.action === 'parseTextOrder') {
      const ss = getSS_();
      return respond_(parseTextOrder_(body.text || '', getNameMap_(ss)));
    }
    return respond_({ ok: false, error: 'unknown action: ' + body.action });
  } catch (err) {
    return respond_({ ok: false, error: err.message });
  }
}

// 把上傳的截圖存到 Drive 資料夾，回傳可分享的連結陣列。
// 存失敗（例如還沒授權 Drive）不擋流程，回空陣列就好，反正這只是輔助查證用的附加功能。
function saveScreenshotsToDrive_(images) {
  try {
    const it = DriveApp.getFoldersByName(SCREENSHOT_DRIVE_FOLDER);
    const folder = it.hasNext() ? it.next() : DriveApp.createFolder(SCREENSHOT_DRIVE_FOLDER);
    const ts = Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'Asia/Taipei', 'yyyyMMdd_HHmmss');
    return images.map(function (img, i) {
      const ext = (img.mimeType || '').indexOf('png') >= 0 ? 'png' : 'jpg';
      const bytes = Utilities.base64Decode(img.data);
      const blob = Utilities.newBlob(bytes, img.mimeType, 'uber_' + ts + '_' + (i + 1) + '.' + ext);
      const file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      return file.getUrl();
    });
  } catch (e) {
    Logger.log('截圖存 Drive 失敗（不影響辨識結果，只是不會有截圖連結）：' + e.message);
    return [];
  }
}

const ORDER_JSON_SCHEMA_HINT =
  '用以下 JSON 格式回答，不要加任何其他文字或說明：' +
  '{"subtotal":0,"deliveryFee":0,"serviceFee":0,"discount":0,"total":0,' +
  '"people":[{"name":"","amt":0}]}';

// 呼叫 Gemini 把 parts(文字提示+圖片，或純文字)解析成訂單結構化資料。
// 回傳 { ok, delivery, discount, people:[{ubName,amt,mappedName}], receipt } 或 { ok:false, error }。
// 圖片辨識(parseUberOrder_)、文字訂單辨識(parseTextOrder_)共用這個函式，只是 parts 內容不同。
function callGeminiForOrder_(parts, nameMap) {
  const apiKey = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!apiKey) return { ok: false, error: 'GEMINI_API_KEY 尚未設定（去 Script Properties 加一組）' };

  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + GEMINI_MODEL +
    ':generateContent?key=' + encodeURIComponent(apiKey);
  const payload = JSON.stringify({
    contents: [{ parts: parts }],
    generationConfig: { responseMimeType: 'application/json' }
  });

  // 503(伺服器忙)/429(超過速率限制) 是暫時性問題，重試幾次通常就過了，不用讓使用者手動重試
  let resp;
  // 只在 GAS 內部小重試一下就好，不要撐太久——Worker 打 GAS 這段是 Cloudflare 的 subrequest，
  // 撐太久(之前設到快49秒)反而會被 Cloudflare 自己判定逾時、回 524，比不重試更難查。
  // 真正撐過持續性尖峰的重試交給前端做「整個請求重打」，見 public/index.html 的 fetchWithRetry_。
  const retryDelays = [1500, 3000];
  for (let attempt = 0; ; attempt++) {
    resp = UrlFetchApp.fetch(url, {
      method: 'post',
      contentType: 'application/json',
      muteHttpExceptions: true,
      payload: payload
    });
    const code = resp.getResponseCode();
    if (code === 200) break;
    const retryable = (code === 503 || code === 429) && attempt < retryDelays.length;
    if (!retryable) {
      return { ok: false, error: 'Gemini 呼叫失敗 (' + code + ')：' + resp.getContentText().slice(0, 300) };
    }
    Utilities.sleep(retryDelays[attempt]);
  }

  var parsed;
  try {
    const data = JSON.parse(resp.getContentText());
    const text = data.candidates[0].content.parts[0].text;
    parsed = JSON.parse(text);
  } catch (e) {
    return { ok: false, error: '辨識結果解析失敗，請重新試一次' };
  }

  const delivery = round2_((parsed.deliveryFee || 0) + (parsed.serviceFee || 0));
  const discount = round2_(parsed.discount || 0);
  const people = (parsed.people || [])
    .filter(function (p) { return p.name && p.amt; })
    .map(function (p) {
      return {
        ubName: p.name,
        amt: round2_(p.amt),
        mappedName: (nameMap && nameMap[cleanName_(p.name)]) || ''
      };
    });

  return {
    ok: true,
    delivery: delivery,
    discount: discount,
    people: people,
    receipt: { subtotal: parsed.subtotal || 0, total: parsed.total || 0 }
  };
}

function parseUberOrder_(images, nameMap) {
  if (!images.length) return { ok: false, error: '沒有收到圖片' };

  const parts = [{
    text: '這是 Uber Eats 團體訂購「結帳」畫面的截圖（可能有多張，是同一筆訂單滑動後的不同段落）。' +
      '請讀出每個人的姓名，以及他點的品項金額加總（同一人有多個品項就加總成一個數字）。' +
      '同時讀出整筆訂單的：小計 subtotal、外送費 deliveryFee、服務費/費用 serviceFee、' +
      '折扣（例如 Uber One 點數折扣）discount、總計 total —— 這些金額都要是「實際折扣後」的數字，' +
      '不要用被劃掉的原價。只回報你在截圖中能清楚讀到的資料，看不到的欄位填 0，不要自己編造姓名或金額。' +
      ORDER_JSON_SCHEMA_HINT
  }];
  images.forEach(function (img) {
    parts.push({ inline_data: { mime_type: img.mimeType, data: img.data } });
  });

  const result = callGeminiForOrder_(parts, nameMap);
  if (!result.ok) return result;

  const screenshotLinks = saveScreenshotsToDrive_(images);
  result.screenshotLink = screenshotLinks.join(', ');
  return result;
}

// 解析像 LINE 群組貼的文字飲料/餐點訂單，例如：
//   小妤：紅宣烏龍微微$35
//   鴨子：波霸紅茶拿鐵一分糖微冰$65
// 每行「姓名 + 品項/備註 + 金額」，格式不完全一致(冒號全半形、有無空白、$在前後)都交給 Gemini 判斷，
// 不用自己寫正則表達式硬猜格式。
function parseTextOrder_(text, nameMap) {
  if (!text || !text.trim()) return { ok: false, error: '沒有收到文字內容' };

  const parts = [{
    text: '這是同事在 LINE 群組貼的文字訂單，通常一行一個人，格式類似「姓名：品項備註$金額」' +
      '（冒號可能是全形或半形、金額符號$可能在前面或後面、也可能沒有$符號，品項與備註可以忽略，只要金額）。' +
      '請讀出每個人的姓名和他要付的金額。這種文字訂單通常沒有外送費/折扣，deliveryFee/discount/serviceFee 都填 0 即可，' +
      'subtotal/total 填全部人的金額加總。只回報看得懂的資料，看不懂的欄位或行就跳過，不要自己編造姓名或金額。' +
      ORDER_JSON_SCHEMA_HINT +
      '\n\n以下是要解析的文字：\n' + text
  }];

  return callGeminiForOrder_(parts, nameMap);
}

function round2_(n) { return Math.round((parseFloat(n) || 0) * 100) / 100; }
